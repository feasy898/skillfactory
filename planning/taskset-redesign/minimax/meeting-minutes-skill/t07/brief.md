## 4 可执行断言脚本（裁判复跑用）

```bash
cd <workdir> && python - <<'PY'
import json,re
from docx import Document
d=Document('<outdir>/纪要.docx')
paras=[p.text for p in d.paragraphs]
full='\n'.join(paras)
print('SEC', {s: paras.count(s) for s in ('一、决议事项','二、待办事项','三、风险与关注')})
print('DEC_NUMS', re.findall(r'^(\d+)\. 【', '\n'.join(p for p in paras if p.startswith(('1.','2.','3.'))), re.M))
print('TABLE_ROWS', len(d.tables[0].rows)-1)
print('SUMMARY', json.load(open('<outdir>/summary.json',encoding='utf-8')))
PY
```
